\noindent Two countries $i\in\{US,R\}$, three sectors $j$, years $t=1,\dots,T$.
%=======================================================
\section*{ Baseline }
%=======================================================
\st{1. Value added.}
\begin{align}
  VA_{i,t}=\sum_j\beta^j_iY^j_{i,t}
\end{align}
\st{2. Consumption expenditure and the investment rate.}
\begin{align}
  E^C_{i,t}=VA_{i,t}-E^X_{i,t}-NX_{i,t},
  \qquad
  x_{i,t}=E^X_{i,t}/VA_{i,t}
\end{align}
\st{3. Consumption shares.}
\begin{align}
  s^j_{i,t}=\frac{E^j_{i,t}-\sum_m\nu^{m,j}_i\big(1-\beta^m_i\big)Y^m_{i,t}
  -\upsilon^j_iE^X_{i,t}}{E^C_{i,t}}
\end{align}
\st{4. Phitilde, how much consumption a dollar buys.}
\begin{align}
  \widetilde{\phi}_{i,t}=\Big(\sum_j\varepsilon^js^j_{i,t}\Big)^{-1}
\end{align}
\st{5. World interest rate,} for $t=1,\dots,T-1$.
\begin{align}
  R_{t+1}=\frac{1}{\delta}\cdot\frac{E^C_{R,t+1}}{E^C_{R,t}}\cdot
  \frac{L_{R,t}}{L_{R,t+1}}\cdot\frac{\widetilde{\phi}_{R,t}}{\widetilde{\phi}_{R,t+1}}
\end{align}
\st{6. The wedge,} for $t=1,\dots,T-1$. ROW's is one.
\begin{align}
  \omega_{US,t}=\frac{L_{US,t+1}}{L_{US,t}}\cdot
  \frac{\widetilde{\phi}_{US,t+1}}{\widetilde{\phi}_{US,t}}\cdot\delta R_{t+1}\cdot
  \frac{E^C_{US,t}}{E^C_{US,t+1}}
\end{align}
\st{7. Bonds.} Start from $B_{i,1}=0$.
\begin{align}
  B_{i,t+1}=R_tB_{i,t}+NX_{i,t}
\end{align}
%=======================================================
\section*{Inner Loop (static year)}
%=======================================================
\noindent Input: year $t$ and US net exports $NX'_{US,t}$
$NX'_{R,t}=-NX'_{US,t}$.

\st{8. Guess the change in the US wage,} $\hd{w}_{US}$. Set $\hd{w}_R=1$. 

\st{9. Unit costs and prices.} Start at $\hd{P}=1$ and alternate the two until they stop moving.
\begin{align}
  \hd{c}^j_i=\big(\hd{w}_i\big)^{\beta^j_i}\prod_m\big(\hd{P}^m_i\big)^{\nu^{j,m}_i(1-\beta^j_i)},
  \qquad
  \hd{P}^j_i=\Big[\sum_h\pi^j_{ih,t}\big(\hd{c}^j_h\big)^{-\theta}\Big]^{-1/\theta}
\end{align}
\st{10. Trade shares.}
\begin{align}
  \pi'^j_{ih}=\pi^j_{ih,t}\left(\frac{\hd{c}^j_h}{\hd{P}^j_i}\right)^{-\theta}
\end{align}
\st{11. Income and spending.}
\begin{align}
  VA'_i=VA_{i,t}\hd{w}_i,
  \qquad
  E^{X\prime}_i=x_{i,t}VA'_i,
  \qquad
  E^{C\prime}_i=\big(1-x_{i,t}\big)VA'_i-NX'_i
\end{align}
\st{12. Real consumption.} Solve for $\hd{C}_i$, one country at a time, where
$\hd{E}^C_i=E^{C\prime}_i/E^C_{i,t}$.
\begin{align}
  \hd{C}_i\left[\sum_js^j_{i,t}\big(\hd{P}^j_i\big)^{1-\psi}
  \hd{C}_i^{\,(1-\psi)(\varepsilon^j-1)}\right]^{\frac{1}{1-\psi}}=\hd{E}^C_i
\end{align}
\st{13. New consumption shares.} The bracket above, to the same power, is $\hd{P}^C_i$.
\begin{align}
  s'^j_i=s^j_{i,t}\left(\frac{\hd{P}^j_i}{\hd{P}^C_i}\right)^{1-\psi}
  \hd{C}_i^{\,(1-\psi)(\varepsilon^j-1)}
\end{align}
\st{14. Market clearing.} Six slots, ordered $(a,US),(a,R),(m,US),(m,R),(s,US),(s,R)$.
\begin{align}
  F_{(j,i)}=s'^j_iE^{C\prime}_i+\upsilon^j_iE^{X\prime}_i,
  \quad
  \Gamma_{(j,i),(m,i)}=\nu^{m,j}_i\big(1-\beta^m_i\big),
  \quad
  \Pi'_{(j,i),(j,h)}=\pi'^j_{hi}
\end{align}
\begin{align}
  E'=\big[I-\Gamma\Pi'\big]^{-1}F,
  \qquad
  Y'=\Pi'E'
\end{align}
\st{15. Income residual.} If this is not zero, go back to step 8.
\begin{align}
  g=\frac{\sum_j\beta^j_{US}Y'^j_{US}}{VA_{US,t}\,\hd{w}_{US}}-1
\end{align}
%=======================================================
\section*{ Outter Loop (Dynamic)} 
%=======================================================
\noindent Input: wedge path $\omega_{US,t}$.
\st{16. Guess the whole path} $\{NX'_{US,t}\}_{t=1}^{T}$. Start at the observed one.

\st{17. Every year.} Run inner loop at each $t$, collect $E^{C\prime}_{i,t}$ and the new
shares, then recompute step 4 from those shares.

\st{18. World interest rate,} step 5 again, on the new consumption path.

\st{19. Bonds,} step 7 again, on the guessed net exports and the new interest rates.

\st{20. US saving equation.} For $t=1,\dots,T-1$,
\begin{align}
  \mathcal{R}_t=\ln\frac{E^{C\prime}_{US,t+1}}{E^{C\prime}_{US,t}}
  -\ln\frac{L_{US,t+1}}{L_{US,t}}
  -\ln\frac{\widetilde{\phi}_{US,t+1}}{\widetilde{\phi}_{US,t}}
  -\ln\frac{\delta R'_{t+1}}{\omega_{US,t}},
\end{align}
\st{21. Terminal condition,} so the counterfactual ends where the baseline did.
\begin{align}
  \mathcal{R}_T=\frac{B'_{US,T+1}-B_{US,T+1}}{VA_{US,T}}.
\end{align}
\st{22. Newton solver}
%=======================================================
\section*{Counterfactual}
%=======================================================
\st{23. Run counterfactual once at the wedge from step 6, once with the wedge set to one and compare manufacturing's share of value added}
\begin{align}
  va^j_{i,t}=\frac{\beta^j_iY'^j_{i,t}}{\sum_m\beta^m_iY'^m_{i,t}},
  \qquad
  \text{answer}=1-\frac{\Delta va^{m,\text{CF}}_{US}}{\Delta va^{m,\text{base}}_{US}},
\end{align}
