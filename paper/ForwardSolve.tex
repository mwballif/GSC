\begin{algorithm}[h]
\caption{}
\begin{algorithmic}[1]
\State \textbf{Solve} the terminal steady state.
\State \textbf{Guess} two paths: $\{K_{i,t}\}$ and $\{NX_{i,t}\}$
\Repeat
  \State Solve the static problem in every year.
  \State Accumulate bonds forward.
  \State Check if consumption Euler holds in each year
  \State Chekc if Capital Euler each year, at the recovered $\chi_{i,t}$.
  \State Terminal conditions: steady states.
  \State Update the guess.
\Until{convergence}
\end{algorithmic}
\end{algorithm}

%=======================================================
\section*{1. Solve steady State }
%=======================================================

\st{Consumption Euler}
\begin{align}
1=\frac{\delta R_{ss}}{1+\phi_{i,ss}\big(\bar{B}_i-\bar{B}_i\big)}
\qquad\Longrightarrow\qquad
R_{ss}=\frac{1}{\delta}
\end{align}


\st{Bond NX identity}
\begin{align}
B_{i,t+1}-B_{i,t}=NX_{i,t}+\big(R_t-1\big)B_{i,t}
\qquad\Longrightarrow\qquad
NX_{i,ss}=-\Big(\tfrac{1}{\delta}-1\Big)\bar{B}_i
\end{align}


\st{Capital Euler}
\begin{align}
\frac{P^X_{i,ss}}{\delta}=r_{i,ss}+(1-d)P^X_{i,ss}
\qquad\Longrightarrow\qquad
\frac{r_{i,ss}}{P^X_{i,ss}}=\frac{1}{\delta}-(1-d)
\end{align}

\st{Capital accumulation}
\begin{align}
K_{i,t+1}=X_{i,t}+(1-d)K_{i,t}
\qquad\Longrightarrow\qquad
X_{i,ss}=d\,K_{i,ss}
\end{align}

\section*{2. Static }
%=======================================================
\noindent Input: the year's disturbances, the capital stocks $K_{i,t}$ and $K_{i,t+1}$, and
net exports $NX_{i,t}$.
 
\st{Guess the wage.}
\begin{align}
w_{i,t}
\end{align}


\st{Recover the rental rate from the capital--labor ratio, using
$r_{i,t}K_{i,t}=\varphi_iVA_{i,t}$ and $w_{i,t}L_{i,t}=(1-\varphi_i)VA_{i,t}$.}
\begin{align}
\frac{w_{i,t}}{r_{i,t}}=\frac{1-\varphi_i}{\varphi_i}\cdot\frac{K_{i,t}}{L_{i,t}}
\end{align}


\st{Obtain unit costs and solve for sectoral prices via system of equations}
\begin{align}
c^j_{i,t}=\varkappa^j_i\Big[\big(r_{i,t}^{\varphi_i}w_{i,t}^{1-\varphi_i}\big)^{\beta^j_i}
\prod_{k\in\mathcal{J}}\big(P^k_{i,t}\big)^{\nu^{j,k}_i(1-\beta^j_i)}\Big],
\qquad
P^j_{i,t}=\Gamma\big[\Phi^j_{i,t}\big]^{-1/\theta},
\quad
\Phi^j_{i,t}=\sum_h T^j_{h,t}\big(c^j_{h,t}\tau^j_{ih,t}\big)^{-\theta}
\end{align}


\st{Construct trade shares.}
\begin{align}
\pi^j_{ih,t}=\frac{T^j_{h,t}\big(c^j_{h,t}\tau^j_{ih,t}\big)^{-\theta}}{\Phi^j_{i,t}}
\end{align}


\st{Investment from the guessed capital stocks}
\begin{align}
X_{i,t}=K_{i,t+1}-(1-d)K_{i,t},
\qquad
P^X_{i,t}=\Big[\sum_j\mu^{x,j}_{i,t}\big(P^j_{i,t}\big)^{1-\zeta}\Big]^{\frac{1}{1-\zeta}},
\qquad
E^X_{i,t}=P^X_{i,t}X_{i,t}
\end{align}


\st{Consumption expenditure is what is left of value added.}
\begin{align}
VA_{i,t}=w_{i,t}L_{i,t}+r_{i,t}K_{i,t},
\qquad
E^C_{i,t}=VA_{i,t}-E^X_{i,t}-NX_{i,t}
\end{align}


\st{Solve for real consumption. $P^C$ depends on $C_{i,t}/L_{i,t}$, so this is implicit.}
\begin{align}
E^C_{i,t}=P^C_{i,t}C_{i,t},
\qquad
P^C_{i,t}=\Big[\sum_j\mu^{c,j}_{i,t}
\big(C_{i,t}/L_{i,t}\big)^{(1-\psi)(\varepsilon^j-1)}
\big(P^j_{i,t}\big)^{1-\psi}\Big]^{\frac{1}{1-\psi}}
\end{align}


\st{Sectoral final demands.}
\begin{align}
s^j_{i,t}=\mu^{c,j}_{i,t}\left(\frac{P^j_{i,t}}{P^C_{i,t}}\right)^{1-\psi}
\left(\frac{C_{i,t}}{L_{i,t}}\right)^{(1-\psi)(\varepsilon^j-1)},
\qquad
x^j_{i,t}=\mu^{x,j}_{i,t}\left(\frac{P^j_{i,t}}{P^X_{i,t}}\right)^{1-\zeta}
\end{align}


\st{Build expenditure from its three uses and output from the trade shares. Iterate the two
until they stop moving.}
\begin{align}
E^j_{i,t}=s^j_{i,t}E^C_{i,t}+x^j_{i,t}E^X_{i,t}
+\sum_{m\in\mathcal{J}}\big(1-\beta^m_i\big)\nu^{m,j}_iY^m_{i,t},
\qquad
Y^j_{i,t}=\sum_h\pi^j_{hi,t}E^j_{h,t}
\end{align}


\st{Sectoral net exports.}
\begin{align}
NX^j_{i,t}=Y^j_{i,t}-E^j_{i,t}
\end{align}
 
\st{ If this does not hold, go back and update the wage.}
\begin{align}
\sum_{j\in\mathcal{J}}NX^j_{i,t}=NX_{i,t}
\end{align}
