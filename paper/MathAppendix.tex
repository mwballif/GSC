\section*{Part A: Derivations of Equilibrium Conditions}
%=======================================================

%-------------------------------------------------------
\subsection*{Households: Setup, Nonhomothetic CES Aggregator}
%-------------------------------------------------------

Following CLM (2017), the household CES aggregator over goods $j \in \mathcal{J}$ satisfies
\begin{align}
  \sum_{j \in \mathcal{J}} \left(\mu^{j}_{i,t}\right)^{\frac{1}{\psi}}
  \left(\frac{C_{i,t}}{L_{i,t}}\right)^{\frac{1 - \psi}{\psi}\varepsilon^j}
  \left(\frac{C^j_{i,t}}{L_{i,t}}\right)^{\frac{\psi - 1}{\psi}} = 1, \label{eq:aggregator}
\end{align}
with $\psi > 0$, $\mu^j_{i,t} > 0$, and $\sum_{j \in \mathcal{J}} e^j = 1$.

%-------------------------------------------------------
\subsection*{Households: Cost Minimization}
%-------------------------------------------------------

The household solves
\begin{align}
  \min_{C^j_{i,t}} \sum_{j \in \mathcal{J}} P^j_{i,t} C^j_{i,t}
  \quad \text{s.t.} \quad \eqref{eq:aggregator}. \label{eq:costmin}
\end{align}

\textit{FOC for $C^j_{i,t}$:}
\begin{align}
  P^j_{i,t} &= \lambda \left(\mu^j_{i,t}\right)^{\frac{1}{\psi}}
               \left(\frac{C_{i,t}}{L_{i,t}}\right)^{\frac{1 - \psi}{\psi}\varepsilon^j}
               \frac{\psi-1}{\psi}
               \left(\frac{C^j_{i,t}}{L_{i,t}}\right)^{-\frac{1}{\psi}}
               \frac{1}{L_{i,t}} \label{eq:foc}
\end{align}

\textit{Relative demand ratio} of two sectors ( $j$ and $j'$):
\begin{align}
  \frac{P^j_{i,t}}{P^{j'}_{i,t}}
  &= \left(\frac{\mu^j_{i,t}}{\mu^{j'}_{i,t}}\right)^{\frac{1}{\psi}}
     \left(\frac{C_{i,t}}{L_{i,t}}\right)^{\frac{1-\psi}{\psi}(\varepsilon^j - \varepsilon^{j'})}
     \left(\frac{C^j_{i,t}}{C^{j'}_{i,t}}\right)^{-\frac{1}{\psi}} \label{eq:ratio}
\end{align}

%-------------------------------------------------------
\subsection*{Households: Demand Expressions}
%-------------------------------------------------------

Solving \eqref{eq:ratio} for $C^j_{i,t}$:
\begin{align}
  \left(C^j_{i,t}\right)^{\frac{1}{\psi}}
  &= \left(\frac{\mu^j_{i,t}}{\mu^{j'}_{i,t}}\right)^{\frac{1}{\psi}}
     \left(\frac{C_{i,t}}{L_{i,t}}\right)^{\frac{1-\psi}{\psi}(\varepsilon^j - \varepsilon^{j'})}
     \left(\frac{P^{j'}_{i,t}}{P^j_{i,t}}\right)
     \left(C^{j'}_{i,t}\right)^{\frac{1}{\psi}} \label{eq:solve1}\\
  \Rightarrow\quad
  \frac{C^j_{i,t}}{L_{i,t}}
  &= \left(\frac{\mu^j_{i,t}}{\mu^{j'}_{i,t}}\right)
     \left(\frac{C_{i,t}}{L_{i,t}}\right)^{(1-\psi)(\varepsilon^j - \varepsilon^{j'})}
     \left(\frac{P^{j'}_{i,t}}{P^j_{i,t}}\right)^{\psi}
     \frac{C^{j'}_{i,t}}{L_{i,t}} \label{eq:demand-raw}
\end{align}

\textit{Plugging \eqref{eq:demand-raw} into the constraint \eqref{eq:aggregator} we have:}
\begin{align*}
    \sum_{j \in \mathcal{J}}
    \left(\mu^j_{i,t}\right)^{\frac{1}{\psi}}
    \left(\frac{C_{i,t}}{L_{i,t}}\right)^{\frac{1-\psi}{\psi}\varepsilon^j}
    \left( \left(\frac{\mu^j_{i,t}}{\mu^{j^{0}}_{i,t}}\right)
     \left( \frac{C_{i,t}}{L_{i,t}}\right)^{(1-\psi)(\varepsilon^j - \varepsilon^{j^{0}})} \left( \frac{P^{j^{0}}_{i,t}}{P^j_{i,t}}\right)^{\psi}
     \frac{C^{j^{0}}_{i,t}}{L_{i,t}} \right)^{\frac{\psi-1}{\psi}} & = 1 \\
     \left(\frac{1}{\mu^{j^{0}}_{i,t}}\right)^{\frac{\psi-1}{\psi}}
     \left( \frac{C_{i,t}}{L_{i,t}}\right)^{\frac{(\psi-1)^{2}}{\psi}\varepsilon^{j^{0}}}
     \left( \frac{P^{j^{0}}_{i,t}}{1}\right)^{\psi-1}
     \left( \frac{C^{j^{0}}_{i,t}}{L_{i,t}} \right)^{\frac{\psi-1}{\psi}}
     \sum_{j \in \mathcal{J}}
     \left(\mu^j_{i,t}\right)^{\frac{1}{\psi}}
     \left(\frac{C_{i,t}}{L_{i,t}}\right)^{\frac{1-\psi}{\psi}\varepsilon^j}
     \left( \left(\frac{\mu^j_{i,t}}{1}\right)
     \left( \frac{C_{i,t}}{L_{i,t}}\right)^{(1-\psi)\varepsilon^j} \left( \frac{1}{P^j_{i,t}}\right)^{\psi} \right)^{\frac{\psi-1}{\psi}} & = 1 \\
     \left(\frac{1}{\mu^{j^{0}}_{i,t}}\right)^{\frac{\psi-1}{\psi}}
     \left( \frac{C_{i,t}}{L_{i,t}}\right)^{\frac{(\psi-1)^{2}}{\psi}\varepsilon^{j^{0}}}
     \left( \frac{P^{j^{0}}_{i,t}}{1}\right)^{\psi-1}
     \left( \frac{C^{j^{0}}_{i,t}}{L_{i,t}} \right)^{\frac{\psi-1}{\psi}}
     \sum_{j \in \mathcal{J}}
     \left(\mu^j_{i,t}\right)^{\frac{1}{\psi} + \frac{\psi-1}{\psi}}
     \left(\frac{C_{i,t}}{L_{i,t}}\right)^{\frac{1-\psi}{\psi}\varepsilon^j - \frac{(1-\psi)^2}{\psi} \varepsilon^j }
     \left( P^j_{i,t} \right)^{1-\psi} & = 1 \\
     \left(\frac{1}{\mu^{j^{0}}_{i,t}}\right)^{\frac{\psi-1}{\psi}}
     \left( \frac{C_{i,t}}{L_{i,t}}\right)^{\frac{(\psi-1)^{2}}{\psi}\varepsilon^{j^{0}}}
     \left( \frac{P^{j^{0}}_{i,t}}{1}\right)^{\psi-1}
     \left( \frac{C^{j^{0}}_{i,t}}{L_{i,t}} \right)^{\frac{\psi-1}{\psi}}
     \sum_{j \in \mathcal{J}}
     \left( \mu^j_{i,t} \right)
     \left(\frac{C_{i,t}}{L_{i,t}}\right)^{(\frac{1}{\psi} - \frac{(1-\psi)}{\psi}) (1-\psi) \varepsilon^j }
     \left( P^j_{i,t} \right)^{1-\psi} & = 1 \\
     \left(\frac{1}{\mu^{j^{0}}_{i,t}}\right)^{\frac{\psi-1}{\psi}}
     \left( \frac{C_{i,t}}{L_{i,t}}\right)^{\frac{(\psi-1)^{2}}{\psi}\varepsilon^{j^{0}}}
     \left( \frac{P^{j^{0}}_{i,t}}{1}\right)^{\psi-1}
     \left( \frac{C^{j^{0}}_{i,t}}{L_{i,t}} \right)^{\frac{\psi-1}{\psi}}
     \sum_{j \in \mathcal{J}}
     \left( \mu^j_{i,t} \right)
     \left(\frac{C_{i,t}}{L_{i,t}}\right)^{ (1-\psi) \varepsilon^j }
     \left( P^j_{i,t} \right)^{1-\psi} & = 1 \\
     \left(\frac{1}{\mu^{j^{0}}_{i,t}}\right)^{\frac{\psi-1}{\psi}}
     \left( \frac{C_{i,t}}{L_{i,t}}\right)^{\frac{(\psi-1)^{2}}{\psi}\varepsilon^{j^{0}} + (1-\psi)}
     \left( \frac{P^{j^{0}}_{i,t}}{1}\right)^{\psi-1}
     \left( \frac{C^{j^{0}}_{i,t}}{L_{i,t}} \right)^{\frac{\psi-1}{\psi}}
     \sum_{j \in \mathcal{J}}
     \left( \mu^j_{i,t} \right)
     \left(\frac{C_{i,t}}{L_{i,t}}\right)^{ (1-\psi) (\varepsilon^j - 1)}
     \left( P^j_{i,t} \right)^{1-\psi} & = 1 \\
     \left(\mu^{j^{0}}_{i,t}\right)^{\frac{1}{\psi}}
     \left( \frac{C_{i,t}}{L_{i,t}}\right)^{\frac{-(\psi-1)\varepsilon^{j^{0}} + \psi}{\psi}}
     \left( P^{j^{0}}_{i,t} \right)^{-1}
     \left( \frac{C^{j^{0}}_{i,t}}{L_{i,t}} \right)^{-\frac{1}{\psi}}
     \left[
     \sum_{j \in \mathcal{J}}
     \left( \mu^j_{i,t} \right)
     \left(\frac{C_{i,t}}{L_{i,t}}\right)^{ (1-\psi) (\varepsilon^j - 1)}
     \left( P^j_{i,t} \right)^{1-\psi} \right]^{\frac{1}{1-\psi}} & = 1
     % \sum_{j \in \mathcal{J}}
     % \left(\mu^j_{i,t}\right)^{\frac{1}{\psi}}
     % \left(\frac{C_{i,t}}{L_{i,t}}\right)^{\frac{1-\psi}{\psi}\varepsilon^j}
     % \left(\frac{\mu^j_{i,t}}{\mu^{j^0}_{i,t}}\right)^{\frac{\psi-1}{\psi}}
     % \left(\frac{C_{i,t}}{L_{i,t}}\right)^{-\frac{(1-\psi)^2}{\psi}(\varepsilon^j - \varepsilon^{j^0})}
     % \left(\frac{P^{j^0}_{i,t}}{P^j_{i,t}}\right)^{\psi-1}
     % \left(\frac{C^{j^0}_{i,t}}{L_{i,t}}\right)^{\frac{\psi-1}{\psi}} & = 1
     % \label{eq:plug2}
\end{align*}

Define the aggregate consumption price index, $P_{i,t}^C$, as follows:
\begin{align}
    P^C_{i,t} \equiv \left[
     \sum_{j \in \mathcal{J}}
     \left( \mu^j_{i,t} \right)
     \left(\frac{C_{i,t}}{L_{i,t}}\right)^{ (1-\psi) (\varepsilon^j - 1)}
     \left( P^j_{i,t} \right)^{1-\psi} \right]^{\frac{1}{1-\psi}}. \label{eq:priceindex}
\end{align}

Then,
\begin{align*}
    \left(\mu^{j^{0}}_{i,t}\right)^{\frac{1}{\psi}}
     \frac{P_{i,t}^{C}}{P^{j^{0}}_{i,t}} & = \left( \frac{C^{j^{0}}_{i,t}}{L_{i,t}} \right)^{\frac{1}{\psi}} \left( \frac{C_{i,t}}{L_{i,t}}\right)^{\frac{(\psi-1)\varepsilon^{j^{0}} - \psi}{\psi}}
\end{align*}


Solving for $\frac{C^{j^0}_{i,t}}{L_{i,t}}$ we have
\begin{align}
  \left(\frac{C^{j^0}_{i,t}}{L_{i,t}}\right)^\frac{1}{\psi} 
  & = \left(\mu^{j^{0}}_{i,t}\right)^{\frac{1}{\psi}}
     \frac{P_{i,t}^{C}}{P^{j^{0}}_{i,t}} 
     \left( \frac{C_{i,t}}{L_{i,t}}\right)^{\frac{(1- \psi)\varepsilon^{j^{0}} + \psi}{\psi}}
  \label{eq:demand-final-j0}
\end{align}


\begin{align}
  \frac{C^{j^0}_{i,t}}{L_{i,t}}
  & = \mu^{j^{0}}_{i,t}
     \left(\frac{P_{i,t}^{C}}{P^{j^{0}}_{i,t}}\right)^{\psi}
     \left( \frac{C_{i,t}}{L_{i,t}}\right)^{(1- \psi)\varepsilon^{j^{0}} + \psi}
  \label{eq:demand-final}
\end{align}

\subsection*{Households: Aggregate Price Index}
%-------------------------------------------------------

Multiplying by prices ($P^j_{i,t}$) and summing over each sector we solve for total expenditure per worker:
\begin{align}
  \frac{1}{L_{i,t}}\sum_{j \in \mathcal{J}} P^j_{i,t} C^j_{i,t}
  &= \sum_{j \in \mathcal{J}} \mu^j_{i,t}
     \left(\frac{C_{i,t}}{L_{i,t}}\right)^{\varepsilon^j}
     \left(P^j_{i,t}\right)^{1-\psi}
     \left(P^C_{i,t}\right)^{\psi} \label{eq:sum1}\\
  &= \left(P^C_{i,t}\right)^{\psi}
     \cdot \frac{C_{i,t}}{L_{i,t}}
     \cdot \underbrace{\left[\sum_{j \in \mathcal{J}} \mu^j_{i,t}
       \left(\frac{C_{i,t}}{L_{i,t}}\right)^{\varepsilon^j-1}
       \left(P^j_{i,t}\right)^{1-\psi}\right]}_{=\,(P^C_{i,t})^{1-\psi}} \label{eq:sum2}\\
  &= \left(P^C_{i,t}\right)^{\psi}
     \cdot \frac{C_{i,t}}{L_{i,t}}
     \cdot \left(P^C_{i,t}\right)^{1-\psi} \\
  &=  P^C_{i,t} \cdot \frac{C_{i,t}}{L_{i,t}} \label{eq:sum3} 
\end{align}

Therefore $E^C_{i,t} \equiv \sum_{j} P^j_{i,t} C^j_{i,t} = P^C_{i,t}\, C_{i,t}$.

%-------------------------------------------------------
\subsection*{Households: Sectoral Final Consumption Expenditure Shares}
%-------------------------------------------------------
Dividing sectoral expenditure by total we derive consumption expenditure shares:
Multiply by sectoral price ($P^j_{i,t}$)
\begin{align}
    \frac{P^j_{i,t} \cdot \frac{C^j_{i,t}}{L_{i,t}}}{P^C_{i,t} \cdot \frac{C_{i,t}}{L_{i,t}}}
        &= \mu^j_{i,t}
        \left(\frac{C_{i,t}}{L_{i,t}}\right)^{\varepsilon^j}
        \left(P^{j}_{i,t}\right)^{-\psi}
        \left(P^C_{i,t}\right)^\psi
        \left(P^C_{i,t}\right)^{-1}
        \left(\frac{C_{i,t}}{L_{i,t}}\right)^{-1} \\
  \Rightarrow\quad
  s^j_{i,t}      &= \mu^j_{i,t} \left(\frac{P^j_{i,t}}{P^C_{i,t}}\right)^{1-\psi}
        \left(\frac{C_{i,t}}{L_{i,t}}\right)^{\varepsilon^j -1} \label{eq:cons-shares}
\end{align}

\subsection*{Households: Investment CES Aggregator and Demand System}
%-------------------------------------------------------

Investment has the same CES structure as consumption, but \emph{homothetic}:
set $\varepsilon^j = 1$ for all $j$ in \eqref{eq:aggregator} and relabel
$(\mu^j_{i,t}, \psi) \to (\upsilon^j_{i}, \zeta)$:
\begin{align}
  \sum_{j \in \mathcal{J}}
  \left(\upsilon^j_{i}\right)^{\frac{1}{\zeta}}
  \left(\frac{X_{i,t}}{L_{i,t}}\right)^{\frac{1-\zeta}{\zeta}}
  \left(\frac{X^j_{i,t}}{L_{i,t}}\right)^{\frac{\zeta-1}{\zeta}} = 1. \label{eq:investagg}
\end{align}
Unlike the nonhomothetic case, \eqref{eq:investagg} can be solved explicitly for the aggregate:
\begin{align}
  X_{i,t}
  = \left[\sum_{j \in \mathcal{J}}
     \left(\upsilon^j_{i}\right)^{\frac{1}{\zeta}}
     \left(X^j_{i,t}\right)^{\frac{\zeta-1}{\zeta}}
     \right]^{\frac{\zeta}{\zeta-1}}. \label{eq:investfinal}
\end{align}

The household minimizes $\sum_{j} P^j_{i,t} X^j_{i,t}$ subject to \eqref{eq:investagg}.

\textit{FOC for $X^j_{i,t}$:}
\begin{align}
  P^j_{i,t} = \lambda \left(\upsilon^j_{i}\right)^{\frac{1}{\zeta}}
  \left(\frac{X_{i,t}}{L_{i,t}}\right)^{\frac{1-\zeta}{\zeta}}
  \frac{\zeta - 1}{\zeta}
  \left(\frac{X^j_{i,t}}{L_{i,t}}\right)^{-\frac{1}{\zeta}}
  \frac{1}{L_{i,t}}. \label{eq:invfoc}
\end{align}

Dividing the FOCs for $j$ and $j'$:
\begin{align}
  \frac{P^j_{i,t}}{P^{j'}_{i,t}}
  = \left(\frac{\upsilon^j_{i}}{\upsilon^{j'}_{i}}\right)^{\frac{1}{\zeta}}
    \left(\frac{X^j_{i,t}}{X^{j'}_{i,t}}\right)^{-\frac{1}{\zeta}}. \label{eq:invratio}
\end{align}

The same steps as \eqref{eq:demand-raw}--\eqref{eq:demand-final} (solve
\eqref{eq:invratio} for $X^j_{i,t}$, substitute into \eqref{eq:investagg}, solve),
now with all income elasticities equal to one, give the sectoral demands and price index:
\begin{align}
  \frac{X^j_{i,t}}{L_{i,t}} = \upsilon^j_{i}
  \left(\frac{X_{i,t}}{L_{i,t}}\right)
  \left(P^j_{i,t}\right)^{-\zeta}
  \left(P^X_{i,t}\right)^{\zeta},
  \qquad
  P^X_{i,t} \equiv \left[\sum_{j\in\mathcal{J}} \upsilon^j_{i}
  \left(P^j_{i,t}\right)^{1-\zeta}\right]^{\frac{1}{1-\zeta}}. \label{eq:invdemand}
\end{align}
Multiplying \eqref{eq:invdemand} by $P^j_{i,t}$ and summing over $j$:
$E^X_{i,t} \equiv \sum_{j} P^j_{i,t}X^j_{i,t} = P^X_{i,t} X_{i,t}$. Dividing
\eqref{eq:invdemand} (times $P^j_{i,t}$) by $E^X_{i,t}$ gives the investment shares:
\begin{align}
  x^j_{i,t} \equiv \frac{P^j_{i,t}X^j_{i,t}}{P^X_{i,t}X_{i,t}}
  = \upsilon^j_{i}\left(\frac{P^j_{i,t}}{P^X_{i,t}}\right)^{1-\zeta}. \label{eq:invshares}
\end{align}

\subsection*{Firms: Unit Costs of Tradable Goods}
%-------------------------------------------------------

Technology for tradable good $\omega^j \in [0,1]$:
\begin{align}
  q^j_{i,t}\left(\omega^j\right) = x^j_{i,t}\left(\omega^j\right)
  \left[k^{\varphi_i}\, l^{1-\varphi_i}\right]^{\beta^j_i}
  \left[\prod_{m\in\mathcal{J}} \left(D^{j,m}\right)^{\nu^{j,m}_i}\right]^{1-\beta^j_i},
  \qquad \sum_{m\in\mathcal{J}}\nu^{j,m}_i = 1. \label{eq:tech}
\end{align}
A producer with efficiency draw $x^j_{i,t}(\omega^j)$ has unit cost
$c^j_{i,t}/x^j_{i,t}(\omega^j)$, where $c^j_{i,t}$ is the cost of one unit of the input bundle
$F \equiv \left[k^{\varphi_i} l^{1-\varphi_i}\right]^{\beta^j_i}
\left[\prod_{m} \left(D^{j,m}\right)^{\nu^{j,m}_i}\right]^{1-\beta^j_i}$:
\begin{align}
  c^j_{i,t} = \min_{k,\,l,\,\{D^{j,m}\}}\;
  r_{i,t}k + w_{i,t}l + \sum_{m\in\mathcal{J}} P^m_{i,t}D^{j,m}
  \quad \text{s.t.} \quad F = 1. \label{eq:firmmin}
\end{align}

Each FOC, evaluated at $F = 1$, pins down spending on that input as a share of $\lambda$:

\textit{FOC for $k$:}
\begin{align}
  r_{i,t} = \lambda\, \beta^j_i \varphi_i\, \frac{F}{k}
  \quad\Rightarrow\quad
  r_{i,t}k = \lambda\, \beta^j_i \varphi_i \label{eq:foc-k}
\end{align}

\textit{FOC for $l$:}
\begin{align}
  w_{i,t} = \lambda\, \beta^j_i (1-\varphi_i)\, \frac{F}{l}
  \quad\Rightarrow\quad
  w_{i,t}l = \lambda\, \beta^j_i (1-\varphi_i) \label{eq:foc-l}
\end{align}

\textit{FOC for $D^{j,m}$:}
\begin{align}
  P^m_{i,t} = \lambda\, (1-\beta^j_i)\nu^{j,m}_i\, \frac{F}{D^{j,m}}
  \quad\Rightarrow\quad
  P^m_{i,t}D^{j,m} = \lambda\, (1-\beta^j_i)\nu^{j,m}_i \label{eq:foc-D}
\end{align}

Summing \eqref{eq:foc-k}--\eqref{eq:foc-D}, total cost equals $\lambda$ (the Cobb--Douglas
shares add to one), so $\lambda$ \emph{is} the unit cost:
\begin{align}
  r_{i,t}k + w_{i,t}l + \sum_{m\in\mathcal{J}} P^m_{i,t}D^{j,m}
  = \lambda \Big[\beta^j_i\varphi_i + \beta^j_i(1-\varphi_i)
  + (1-\beta^j_i)\underbrace{\textstyle\sum_m \nu^{j,m}_i}_{=1}\Big]
  = \lambda
  \quad\Rightarrow\quad \lambda = c^j_{i,t}.
\end{align}
The input demands follow from the FOCs:
\begin{align}
  k = \frac{\beta^j_i\varphi_i\, c^j_{i,t}}{r_{i,t}}, \qquad
  l = \frac{\beta^j_i(1-\varphi_i)\, c^j_{i,t}}{w_{i,t}}, \qquad
  D^{j,m} = \frac{(1-\beta^j_i)\nu^{j,m}_i\, c^j_{i,t}}{P^m_{i,t}}. \label{eq:firmdemands}
\end{align}
Substituting \eqref{eq:firmdemands} into $F = 1$:
\begin{align}
  \left[\left(\frac{\beta^j_i\varphi_i\, c^j_{i,t}}{r_{i,t}}\right)^{\varphi_i}
        \left(\frac{\beta^j_i(1-\varphi_i)\, c^j_{i,t}}{w_{i,t}}\right)^{1-\varphi_i}\right]^{\beta^j_i}
  \prod_{m\in\mathcal{J}}\left(\frac{(1-\beta^j_i)\nu^{j,m}_i\, c^j_{i,t}}{P^m_{i,t}}\right)^{(1-\beta^j_i)\nu^{j,m}_i} = 1.
\end{align}
The exponents on $c^j_{i,t}$ sum to one, so:
\begin{align}
  c^j_{i,t} = \kappa^j_i
  \left[\left(\left(r_{i,t}\right)^{\varphi_i}\left(w_{i,t}\right)^{1-\varphi_i}\right)^{\beta^j_i}
  \left(\prod_{k\in\mathcal{J}}\left(P^k_{i,t}\right)^{\nu^{j,k}_i}\right)^{1-\beta^j_i}\right],
  \label{eq:unitcost}
\end{align}
with constant
\begin{align}
  \kappa^j_i \equiv
  \left[\varphi_i^{\varphi_i}(1-\varphi_i)^{1-\varphi_i}\right]^{-\beta^j_i}
  \left(\beta^j_i\right)^{-\beta^j_i}
  \left(1-\beta^j_i\right)^{-(1-\beta^j_i)}
  \prod_{m\in\mathcal{J}}\left(\nu^{j,m}_i\right)^{-(1-\beta^j_i)\nu^{j,m}_i}.
\end{align}

%-------------------------------------------------------
\subsection*{Households: Dynamic Problem}
%-------------------------------------------------------

The household chooses $\left\{\bar{C}_{i,t}, B_{i,t+1}\right\}_{t=0}^{\infty}$ to maximize
\begin{align}
  \sum_{t=0}^{\infty} \delta^t L_{i,t}\, U\!\left(\bar{C}_{i,t}\right)
\end{align}
subject to the sequence of budget constraints
\begin{align}
  E^C_{i,t}\!\left(\bar{C}_{i,t},\{P^j_{i,t}\}_{j\in\mathcal{J}}\right) + P^X_{i,t}X_{i,t} + B_{i,t+1}
  + \tfrac{\phi_{i,t}}{2}\left(B_{i,t+1}-\bar{B}_i\right)^2
  = w_{i,t}L_{i,t} + r_{i,t}K_{i,t} + R_{i,t}B_{i,t} + T_{i,t}. \label{eq:bc}
\end{align}

Lagrangian:
\begin{align}
  \mathcal{L} &= \sum_{t=0}^{\infty} \delta^t L_{i,t}\, U\!\left(\bar{C}_{i,t}\right)
  + \sum_{t=0}^{\infty} \lambda_{i,t}\Big[ w_{i,t}L_{i,t} + r_{i,t}K_{i,t} + R_{i,t}B_{i,t} + T_{i,t} \notag \\
  &\qquad\qquad - E^C_{i,t} - P^X_{i,t}X_{i,t}
  - B_{i,t+1} - \tfrac{\phi_{i,t}}{2}\left(B_{i,t+1}-\bar{B}_i\right)^2 \Big]. \label{eq:lagrangian}
\end{align}

Marginal consumption expenditure. From \eqref{eq:priceindex}, $P^C_{i,t}$ depends on
$\bar{C}_{i,t} \equiv C_{i,t}/L_{i,t}$. Differentiate both sides of \eqref{eq:priceindex}
w.r.t.\ $\bar{C}_{i,t}$ and divide by $(1-\psi)$:
\begin{align}
  \left(P^C_{i,t}\right)^{-\psi}\frac{\partial P^C_{i,t}}{\partial \bar{C}_{i,t}}
  = \sum_{j\in \mathcal{J}}\mu^j_{i,t}\left(\varepsilon^j-1\right)
  \bar{C}_{i,t}^{\,(1-\psi)\left(\varepsilon^j-1\right)-1}\left(P^j_{i,t}\right)^{1-\psi}.
  \label{eq:diff-raw}
\end{align}
The shares \eqref{eq:cons-shares} rearrange to

\end{align}
Summing over $j$ reproduces \eqref{eq:PChat}. With equal income elasticities
$\hd{C}_i^{(1-\psi)(\varepsilon^j-1)}$ is common to all $j$ and cancels against $\hd{P}^C_i$:
this term is the structural change mechanism.

\subsection*{Market Clearing: The Leontief System}
%-------------------------------------------------------
Scaling \eqref{eq:foc-D} to sector output, the intermediate term in \eqref{eq:mc-exp} is
$\sum_m\nu^{m,j}_i\left(1-\beta^m_i\right)Y^m_{i,t}$, with gross output as the base and the
buying sector $m$ transposed relative to \eqref{eq:tech}. Substituting it and eliminating $Y$
with $Y^j_{i,t}=\sum_h\pi^j_{hi,t}E^j_{h,t}$ leaves the expenditures on both sides, but
linearly:
\begin{align}
  E'^j_i=s'^j_iE^{C\prime}_i+\upsilon^j_iE^{X\prime}_i
  +\sum_{m\in\mathcal{J}}\nu^{m,j}_i\left(1-\beta^m_i\right)\sum_h\pi'^m_{hi}E'^m_h.
  \label{eq:Esub}
\end{align}
Stacking the country-sector pairs, with $\Pi'$ mapping expenditure to output and $\Gamma$
output back to input demand,
\begin{align}
  F_{(j,i)}=s'^j_iE^{C\prime}_i+\upsilon^j_iE^{X\prime}_i,
  \quad
  \Gamma_{(j,i),(m,i)}=\nu^{m,j}_i\left(1-\beta^m_i\right),
  \quad
  \Pi'_{(j,i),(j,h)}=\pi'^j_{hi},
\end{align}
\begin{align}
  E'=\left[I-\Gamma\Pi'\right]^{-1}F, \qquad Y'=\Pi'E'. \label{eq:leontief}
\end{align}
$\Gamma$ is block diagonal by country and holds no prices; $\Pi'$ is block diagonal by sector
and moves with the guess. The inverse exists because the column sums of $\Gamma\Pi'$ are
bounded by the intermediate share:
\begin{align}
  \sum_{(j,i)}\left(\Gamma\Pi'\right)_{(j,i),(m,h)}
  = \sum_i\left(1-\beta^m_i\right)\pi'^m_{hi}\underbrace{\sum_j\nu^{m,j}_i}_{=\,1}
  \le \max_i\left(1-\beta^m_i\right)<1. \label{eq:colsums}
\end{align}

\subsection*{Closing the Static Block}
%-------------------------------------------------------
Output implies income through $VA'_i=\sum_j\beta^j_iY'^j_i$, so a correct wage reproduces
itself and the year is a scalar fixed point:
\begin{align}
  g\left(\hd{w}_{US}\right)
  \equiv\frac{\sum_j\beta^j_{US}Y'^j_{US}}{VA_{US,t}\hd{w}_{US}}-1=0. \label{eq:gresid}
\end{align}
The residual falls in the guess: a higher wage raises costs by \eqref{eq:chat} and prices by
\eqref{eq:Phat}, buyers shift away by \eqref{eq:pinew}, and implied income returns below the
guess.

\textit{The redundant equation.} Summing \eqref{eq:mc-exp} over sectors returns
\eqref{eq:budget-reduced}, and over countries as well makes world output equal world
expenditure identically, hence $\sum_i NX_{i,t}=0$. One of the twenty-four static equations is
implied by the others.

\textit{The numeraire.} The system is homogeneous of degree one in prices, nominal quantities
and $NX$, so it fixes only relative prices. Replacing the redundant equation with
\begin{align}
  \hd{w}_R=1 \label{eq:numeraire}
\end{align}
and using $NX'_{R,t}=-NX'_{US,t}$ leaves $\hd{w}_{US}$ as the only free number in the year.

\subsection*{Closing the Intertemporal Block}
%-------------------------------------------------------
Equation \eqref{eq:euler-exp} holds for $t=1,\dots,T-1$: $T-1$ restrictions on the $T$
unknowns $\{NX'_{US,t}\}_{t=1}^T$. It restricts the tilt of the path, not its level, since
shifting borrowing up in every year leaves every instance satisfied. Part A pins the level by
transversality; truncating at $T$ removes it, so we impose
\begin{align}
  B'_{US,T+1}=B_{US,T+1}. \label{eq:terminal}
\end{align}
Without it the counterfactual can borrow and never repay, and the comparison measures a
windfall. It is a choice, and it binds at the last date, which is where the outcome is
measured, so the answer's sensitivity to the horizon is a diagnostic.

The world interest rate is not an unknown: with $\phi_{ROW,t}=0$ from Part B the rest of the
world's saving equation contains only $R_{t+1}$ and \eqref{eq:worldR} gives it, now on the
counterfactual consumption path.
