\section*{Prelude}
%=======================================================

The model centers on a nonhomothetic CES aggregator over final consumption goods
$j \in \mathcal{J}$:

\[
  \sum_{j \in \mathcal{J}}
  \left(\mu^{c,j}_{i,t}\right)^{\frac{1}{\psi}}
  \left( \frac{C_{i,t}}{L_{i,t}} \right)^{\frac{1-\psi}{\psi}\varepsilon^{j}}
  \left( \frac{C^{j}_{i,t}}{L_{i,t}} \right)^{\frac{\psi-1}{\psi}} = 1,
\]

subject to the sector-specific income elasticities summing to one
($\sum_{j \in \mathcal{J}} \varepsilon^{j} = 1$), the shifters summing to a
country-specific constant ($\sum_{j \in \mathcal{J}} \mu^{c,j}_{i,t} = \kappa_i$),
and the existence of an associated ideal price index $P_{i,t}$. All disturbances
of the model are recovered from the data; the appendix restates the derivations
of the household CES aggregator, the demand system, the unit cost of tradable
goods, and the dynamic household problem in Part A, and shows how each of the
model's disturbances --- the investment shifters, the consumption shifters, the
tradable productivities, the trade costs, the financial friction, and the
investment tax --- is recovered in Part B.

%=======================================================
\section*{Part A: Derivations of Equilibrium Conditions}
%=======================================================
%-------------------------------------------------------
\subsection*{Households: Setup, Nonhomothetic CES Aggregator}
%-------------------------------------------------------
Following CLM (2017), the household CES aggregator over goods $j \in \mathcal{J}$ satisfies
\begin{align}
  \sum_{j \in \mathcal{J}} \left(\mu^{c,j}_{i,t}\right)^{\frac{1}{\psi}}
  \left(\frac{C_{i,t}}{L_{i,t}}\right)^{\frac{1 - \psi}{\psi}\varepsilon^j}
  \left(\frac{C^j_{i,t}}{L_{i,t}}\right)^{\frac{\psi - 1}{\psi}} = 1, \label{eq:aggregator}
\end{align}
with $\psi > 0$, $\mu^{c,j}_{i,t} > 0$, $\sum_{j \in \mathcal{J}} \varepsilon^j = 1$,
and $\sum_{j \in \mathcal{J}}\mu^{c,j}_{i,t} = \kappa_i$ for all $t$. The superscript $c$
distinguishes it from the investment shifter $\mu^{x,j}_{i,t}$ below.

%-------------------------------------------------------
\subsection*{Households: Cost Minimization}
%-------------------------------------------------------
The household minimizes the cost of reaching a given aggregate consumption level
$C_{i,t}$ by choosing sectoral quantities $C^j_{i,t}$ subject to the aggregator
constraint \eqref{eq:aggregator}:
\begin{align}
  \min_{\{ C^j_{i,t} \}} \; \sum_{j \in \mathcal{J}} P^j_{i,t} C^j_{i,t}
  \quad \text{s.t.} \quad
  \sum_{j \in \mathcal{J}} \left(\mu^{c,j}_{i,t}\right)^{\frac{1}{\psi}}
  \left(\frac{C_{i,t}}{L_{i,t}}\right)^{\frac{1 - \psi}{\psi}\varepsilon^j}
  \left(\frac{C^j_{i,t}}{L_{i,t}}\right)^{\frac{\psi - 1}{\psi}} = 1. \label{eq:costmin}
\end{align}
The first-order condition equates the marginal rate of substituting sectoral good
$j$ for the aggregate bundle to its relative price,
\begin{align}
  \frac{\partial C_{i,t}}{\partial C^j_{i,t}} = P^j_{i,t}. \label{eq:foc}
\end{align}
Taking the relevant derivatives, and using the aggregate price index identity
$P_{i,t} C_{i,t} = \sum_{j \in \mathcal{J}} P^j_{i,t} C^j_{i,t}$, one obtains the
share ratio between any two sectors,
\begin{align}
  \frac{P^j_{i,t} C^j_{i,t}}{P^h_{i,t} C^h_{i,t}} =
  \left( \frac{\mu^{c,j}_{i,t}}{\mu^{c,h}_{i,t}} \right)^{\frac{1}{\psi}}
  \left( \frac{C_{i,t}}{L_{i,t}} \right)^{\frac{1 - \psi}{\psi} (\varepsilon^j - \varepsilon^h) }
  \left( \frac{C^j_{i,t}}{C^h_{i,t}} \right)^{\frac{\psi - 1}{\psi}}. \label{eq:ratio}
\end{align}
Setting sector $h = 0$ as the num\'eraire and solving for $C^j_{i,t}$ in terms of $C^{0}_{i,t}$,
\begin{align}
  C^j_{i,t} = C^{0}_{i,t} \left( \frac{P^j_{i,t}}{P^{0}_{i,t}} \right)^{\frac{\psi}{\psi - 1}}
  \left( \frac{\mu^{c,j}_{i,t}}{\mu^{c,0}_{i,t}} \right)^{\frac{1}{\psi - 1}}
  \left( \frac{C_{i,t}}{L_{i,t}} \right)^{\frac{1 - \psi}{\psi} \frac{\psi}{\psi - 1} (\varepsilon^j - \varepsilon^{0})}
  . \label{eq:solve1}
\end{align}
Summing the implied expenditure shares over all sectors and imposing the unit-cost
condition pins down the scale of $C^{0}_{i,t}$, giving the raw demand expression,
\begin{align}
  C^j_{i,t} = \frac{1}{P_{i,t}} \left( \frac{P_{i,t}}{P^j_{i,t}} \right)^{\frac{\psi}{\psi - 1}}
  \mu^{c,j}_{i,t} \left( \frac{C_{i,t}}{L_{i,t}} \right)^{1 - \frac{1 - \psi}{\psi} \varepsilon^j}
  . \label{eq:demand-raw}
\end{align}

%-------------------------------------------------------
\subsection*{Households: Demand Expressions}
%-------------------------------------------------------
The demand system in \eqref{eq:demand-raw} has three useful summation identities:
the sectoral shares sum to one,
\begin{align}
  \sum_{j \in \mathcal{J}} s^j_{i,t} = 1, \label{eq:sum1}
\end{align}
the shifters sum to $\kappa_i$,
\begin{align}
  \sum_{j \in \mathcal{J}} \mu^{c,j}_{i,t} = \kappa_i, \label{eq:sum2}
\end{align}
and the expenditure-weighted elasticities are tied to the aggregate
\begin{align}
  \sum_{j \in \mathcal{J}} s^j_{i,t} \varepsilon^j = 1. \label{eq:sum3}
\end{align}

%-------------------------------------------------------
\subsection*{Households: Aggregate Price Index}
%-------------------------------------------------------
Imposing the requirement that the cost of the aggregate basket equals total
expenditure, $P_{i,t} C_{i,t} = \sum_{j} P^j_{i,t} C^j_{i,t}$, and substituting the
demands \eqref{eq:demand-raw} yields the ideal price index
\begin{align}
  P_{i,t} = \left( \sum_{j \in \mathcal{J}} \mu^{c,j}_{i,t}
  \left( P^j_{i,t} \right)^{\frac{\psi}{\psi - 1}} \right)^{\frac{\psi - 1}{\psi}}
  \left( \frac{C_{i,t}}{L_{i,t}} \right)^{1 - \psi},
\end{align}
which, rewritten in terms of the consumption expenditure level $E^C_{i,t} = P_{i,t} C_{i,t}$,
justifies recovering the sectoral budget shares as
\begin{align}
  s^j_{i,t} \equiv \frac{P^j_{i,t} C^j_{i,t}}{E^C_{i,t}}
  = \left( \frac{\mu^{c,j}_{i,t}}{P_{i,t}^{\frac{\psi}{\psi - 1}}} \right)
  \left( \frac{P^j_{i,t}}{P_{i,t}} \right)^{\frac{\psi}{\psi - 1}}
  \left( \frac{C_{i,t}}{L_{i,t}} \right)^{1 - \psi}. \label{eq:cons-shares}
\end{align}

%-------------------------------------------------------
\subsection*{Households: Investment CES Aggregator and Demand System}
%-------------------------------------------------------
Investment is defined symmetrically with its own shifters $\mu^{x,j}_{i,t}$ and its
own elasticity $\zeta$:
\begin{align}
  X_{i,t} = \left( \sum_{j \in \mathcal{J}} \left(\mu^{x,j}_{i,t}\right)^{\frac{1}{\zeta}}
  \left( X^j_{i,t} \right)^{\frac{\zeta - 1}{\zeta}} \right)^{\frac{\zeta}{\zeta - 1}},
  \label{eq:investagg}
\end{align}
with $\sum_{j \in \mathcal{J}} \mu^{x,j}_{i,t} = \kappa_i$. Cost minimization at the
investment price index $P^{X}_{i,t}$ gives the sectoral investment demands
\begin{align}
  X^j_{i,t} = \mu^{x,j}_{i,t}
  \left( \frac{P^{X}_{i,t}}{P^j_{i,t}} \right)^{\zeta} X_{i,t}, \label{eq:invfoc}
\end{align}
so the price index reads
\begin{align}
  P^{X}_{i,t} = \left( \sum_{j \in \mathcal{J}} \mu^{x,j}_{i,t}
  \left( P^j_{i,t} \right)^{1 - \zeta} \right)^{\frac{1}{1 - \zeta}}, \label{eq:invfinal}
\end{align}
and the investment expenditure shares are
\begin{align}
  s^{x,j}_{i,t} \equiv \frac{P^j_{i,t} X^j_{i,t}}{P^{X}_{i,t} X_{i,t}}
  = \mu^{x,j}_{i,t} \left( \frac{P^{X}_{i,t}}{P^j_{i,t}} \right)^{\zeta - 1}
  , \label{eq:invshares}
\end{align}
with the two auxiliary expressions
\begin{align}
  \left( \frac{P^{X}_{i,t}}{P^j_{i,t}} \right)^{\zeta - 1} = \frac{s^{x,j}_{i,t}}{\mu^{x,j}_{i,t}}
  \quad\text{and}\quad
  \left( \frac{P^{X}_{i,t}}{P^j_{i,t}} \right)^{\zeta}
  = \frac{s^{x,j}_{i,t}}{\mu^{x,j}_{i,t}} \frac{P^{X}_{i,t}}{P^j_{i,t}}. \label{eq:invratio}
\end{align}

%-------------------------------------------------------
\subsection*{Firms: Unit Costs of Tradable Goods}
%-------------------------------------------------------
Sector $j$ produces with nested CES technology over local capital, local labor,
and domestic/intermediate goods:
\begin{align}
  Q^j_{i,t} = A^j_{i,t} \left( K^j_{i,t} \right)^{\theta^j}
  \left( L^j_{i,t} \right)^{1 - \theta^j}
  \left( \sum_{h \in \mathcal{J}} \Gamma^{h,j}_{i,t}
  \left( D^{hj}_{i,t} \right)^{\frac{\eta - 1}{\eta}} \right)^{\frac{\eta}{\eta - 1}},
  \label{eq:tech}
\end{align}
where $\Gamma^{h,j}_{i,t}$ are trade shares, $\eta$ is the Armington elasticity,
and $D^{hj}_{i,t}$ is country $i$'s use of country $h$'s sector-$j$ variety.
Cost minimization delivers the factor demands
\begin{align}
  K^j_{i,t} \propto \frac{\theta^j R_{i,t}}{Q^j_{i,t}}, \qquad
  L^j_{i,t} \propto \frac{(1 - \theta^j) W_{i,t}}{Q^j_{i,t}}, \qquad
  D^{hj}_{i,t} \propto \frac{\Gamma^{h,j}_{i,t} P^{h,j}_{i,t}}{Q^j_{i,t}}, \label{eq:foc-k}
\end{align}
and the unit cost of producing the tradable good
\begin{align}
  \upsilon^{j}_{i,t} = \frac{1}{A^j_{i,t}}
  \left( \frac{R_{i,t}}{\theta^j} \right)^{\theta^j}
  \left( \frac{W_{i,t}}{1 - \theta^j} \right)^{1 - \theta^j}
  \left( \sum_{h \in \mathcal{J}} \Gamma^{h,j}_{i,t}
  \left( P^{h,j}_{i,t} \right)^{1 - \eta} \right)^{\frac{1}{1 - \eta}}, \label{eq:unitcost}
\end{align}

%-------------------------------------------------------
\subsection*{Households: Dynamic Problem}
%-------------------------------------------------------
The household receives labor income, capital-service income, and profits, pays
investment taxes, and holds a one-period bond with friction $\phi_{i,t}$:
\begin{align}
  P_{i,t} C_{i,t} + P^{X}_{i,t} X_{i,t} + B_{i,t+1} \le
  W_{i,t} L_{i,t} + R_{i,t} K_{i,t} + \Pi_{i,t}
  + \left( 1 - \phi_{i,t} \right) R_{t} B_{i,t} - \xi_{i,t} P^{X}_{i,t} X_{i,t},
  \label{eq:bc}
\end{align}
where $\xi_{i,t}$ is the investment tax rate and $R_{t}$ is the common world gross
interest rate,
\begin{align}
  \chi_{i,t} \equiv \frac{1 - R^{\cal P}_{i,t}}{1 - \phi_{i,t}}, \label{eq:chi-def}
\end{align}
and lump-sum rebates distribute any gap:
\begin{align}
  \Xi_{i,t} = \frac{\xi_{i,t} P^{X}_{i,t} X_{i,t}}{1 - \phi_{i,t}}. \label{eq:rebate}
\end{align}
Dividing by the friction and using the resource constraint gives the budget in
expenditure form,
\begin{align}
  E^C_{i,t} + P^{X}_{i,t} X_{i,t} + B_{i,t+1} = W_{i,t} L_{i,t} + R_{i,t} K_{i,t}
  + \Pi_{i,t} + R_{t} B_{i,t}, \label{eq:bc-resource}
\end{align}
The Lagrangian is
\begin{align}
  {\cal L}_t = \beta^t \frac{C^{1 - \sigma}_{i,t}}{1 - \sigma} +
  \Lambda_{i,t} \left[ W_{i,t} L_{i,t} + R_{i,t} K_{i,t} + \Pi_{i,t}
  + R_{t} B_{i,t} - E^C_{i,t} - P^{X}_{i,t} X_{i,t} - B_{i,t+1} \right],
  \label{eq:lagrangian}
\end{align}
with the two envelope conditions
\begin{align}
  \frac{\partial {\cal L}_t}{\partial C_{i,t}} = \beta^t C^{-\sigma}_{i,t},
  \qquad
  \frac{\partial {\cal L}_t}{\partial L_{i,t}} = \beta^t
  \left( 1 - \sigma \right) \frac{C^{-\sigma}_{i,t}}{L_{i,t}}, \label{eq:dEdC}
\end{align}
Collecting the period-$t$ and period-$(t+1)$ terms of the Bellman envelope, the
intratemporal conditions are
\begin{align}
  \frac{\partial C_{i,t}}{\partial S^j_{i,t}} = P^j_{i,t}, \label{eq:lambda-dyn}
\end{align}
and the Euler condition for the bond is
\begin{align}
  1 = \beta R_{t+1} \left( \frac{C_{i,t+1}}{C_{i,t}} \right)^{-\sigma}
  \left( \frac{L_{i,t+1}}{L_{i,t}} \right)^{\sigma - 1}, \label{eq:euler}
\end{align}

%-------------------------------------------------------
\subsection*{Households: Capital Accumulation Euler Equation}
%-------------------------------------------------------
Combining the rental-rate envelope with the investment-technology trade-off gives
the general capital Euler equation:
\begin{align}
  \frac{1}{1 - \xi_{i,t}} = \beta \left( \frac{C_{i,t+1}}{C_{i,t}} \right)^{-\sigma}
  \left( \frac{L_{i,t+1}}{L_{i,t}} \right)^{\sigma - 1}
  \left[ \frac{R_{i,t+1}}{P^{X}_{i,t}} +
  \frac{1 - \delta}{1 - \xi_{i,t+1}} \frac{P^{X}_{i,t+1}}{P^{X}_{i,t}} \right],
  \label{eq:capital-euler-gen}
\end{align}
With no capital adjustment costs and zero depreciation this simplifies to
\begin{align}
  \frac{1}{1 - \xi_{i,t}} = \beta \left( \frac{C_{i,t+1}}{C_{i,t}} \right)^{-\sigma}
  \left( \frac{L_{i,t+1}}{L_{i,t}} \right)^{\sigma - 1} \frac{R_{i,t+1}}{P^{X}_{i,t}},
  \label{eq:capital-euler}
\end{align}
which, after dividing the analogous equation for period $t+1$, yields the
capital-breadth ratio
\begin{align}
  \frac{R_{i,t+1}}{R_{i,t}} =
  \frac{1 - \xi_{i,t}}{1 - \xi_{i,t+1}} \frac{P^{X}_{i,t+1}}{P^{X}_{i,t}}
  \left( \frac{C_{i,t+1}}{C_{i,t}} \right)^{\sigma}
  \left( \frac{L_{i,t+1}}{L_{i,t}} \right)^{1 - \sigma}. \label{eq:capital-euler-ratio}
\end{align}

\section*{Part B: Recovering the Model's Disturbances}
%=======================================================
Every disturbance is a function of observables and of the structural parameters;
each subsection gives the closed-form recovery. Throughout, nominal aggregates
are scaled by the country's population so that per-capita magnitudes enter the
formulas.

%-------------------------------------------------------
\subsection*{Investment Demand Shifters}
%-------------------------------------------------------
Using the investment share equation \eqref{eq:invshares} in logs,
\begin{align}
  \ln s^{x,j}_{i,t} = \ln \mu^{x,j}_{i,t} + \left( \zeta - 1 \right)
  \left( \ln P^{X}_{i,t} - \ln P^j_{i,t} \right), \label{eq:mux}
\end{align}
so the recovered shifters are
\begin{align}
  \mu^{x,j}_{i,t} = s^{x,j}_{i,t}
  \left( \frac{P^{X}_{i,t}}{P^j_{i,t}} \right)^{1 - \zeta}, \label{eq:PXrecover}
\end{align}

%-------------------------------------------------------
\subsection*{Consumption Demand Shifters}
%-------------------------------------------------------
From the consumption share equation \eqref{eq:cons-shares}, define the
expenditure-share-weighted mean of the log price index,
\begin{align}
  \ln P_{i,t} = \sum_{j \in \mathcal{J}} s^j_{i,t} \ln P^j_{i,t}, \label{eq:mc-exp}
\end{align}
and the supply-side expression for the aggregate consumption price is
\begin{align}
  \ln P_{i,t} = \wbar{\ln p^j_{i,t}} - \frac{\psi - 1}{\psi} \ln
  \left( \frac{C_{i,t}}{L_{i,t}} \right), \label{eq:consexp-supply}
\end{align}
where $\wbar{\ln p^j_{i,t}}$ is the expenditure-weighted mean of $\ln P^j_{i,t}$
computed from the estimated model. Matching the demand side,
\begin{align}
  \ln P_{i,t} = \frac{\psi - 1}{\psi} \ln \left( \frac{C_{i,t}}{L_{i,t}} \right)
  - \frac{1}{\psi} \ln \left( \sum_{j \in \mathcal{J}} s^j_{i,t} \right),
  \label{eq:consexp-demand1}
\end{align}
and, using the recovered budget shares,
\begin{align}
  s^j_{i,t} = \sum_{j \in \mathcal{J}} \frac{1}{\psi} s^j_{i,t} \varepsilon^j
  \qquad\Longrightarrow\qquad
  \ln \mu^{c,j}_{i,t} = \psi \left( \ln P^j_{i,t} - \ln P_{i,t} \right)
  + \ln s^j_{i,t}. \label{eq:consexp-demand2}
\end{align}
The consumption aggregation shifters therefore read
\begin{align}
  \mu^{c,j}_{i,t} = s^j_{i,t} \left( \frac{P^j_{i,t}}{P_{i,t}} \right)^{\psi}
  \left( \frac{C_{i,t}}{L_{i,t}} \right)^{1 - \psi}, \label{eq:mu-tradables}
\end{align}

%-------------------------------------------------------
\subsection*{Sectoral Productivities and Trade Costs}
%-------------------------------------------------------
From the firm's cost-minimization FOCs \eqref{eq:foc-D} and the unit cost
\eqref{eq:unitcost}, the terms-of-trade ratio between two countries $h$ and $i$
in sector $j$ is
\begin{align}
  \frac{P^{h,j}_{i,t}}{P^{i,j}_{i,t}} =
  \left( \frac{\Gamma^{i,j}_{i,t}}{\Gamma^{h,j}_{h,t}} \right)^{\frac{1}{1 - \eta}}
  \frac{\upsilon^{j}_{h,t}}{\upsilon^{j}_{i,t}}, \label{eq:Phi-inv}
\end{align}
which, using the gravity trade-share ratio, gives the bilateral share ratio:
\begin{align}
  \frac{\pi^{h,j}_{i,t}}{\pi^{i,j}_{i,t}} =
  \left( \frac{\Gamma^{i,j}_{h,t}}{\Gamma^{h,j}_{h,t}} \right)
  \left( \frac{P^{i,j}_{h,t}}{P^{h,j}_{h,t}} \right)^{1 - \eta}, \label{eq:gravity-ratio}
\end{align}
Setting $h = i$ in the gravity equation \eqref{eq:gravity} to get the home share,
\begin{align}
  \pi^{i,j}_{i,t} = \frac{\Gamma^{i,j}_{i,t} \left( P^{i,j}_{i,t} \right)^{1 - \eta}}
  {\sum_{k} \Gamma^{k,j}_{i,t} \left( P^{k,j}_{i,t} \right)^{1 - \eta}} =
  \frac{1}{1 + \sum_{h \neq i} \frac{\pi^{h,j}_{i,t}}{\pi^{i,j}_{i,t}}}, \label{eq:homeshare}
\end{align}

%-------------------------------------------------------
\subsection*{Financial Friction $\phi_{i,t}$}
%-------------------------------------------------------
The world interest rate comes from the US Euler equation (expenditure form),
\begin{align}
  1 = \beta R_{t+1} \left( \frac{E^C_{US,t+1}}{E^C_{US,t}} \right)^{-\sigma}
  \left( \frac{L_{US,t+1}}{L_{US,t}} \right)^{2\sigma - 1}, \label{eq:euler-exp}
\end{align}
using the balance-of-payments identity for the ROW mirror,
\begin{align}
  B_{R,t+1} = - B_{US,t+1}, \label{eq:bop}
\end{align}
so that bond positions mirror exactly,
\begin{align}
  B_{R,t} = - B_{US,t}, \label{eq:bmirror}
\end{align}
Then the friction is recovered from the ROW Euler equation:
\begin{align}
  \phi_{R,t} = 1 - \frac{R_{t+1}}{\bar{R}_{R,t+1}}, \label{eq:phi-recover}
\end{align}
where $\bar{R}_{R,t+1}$ is the ROW's own rate that would clear its bond market
in the absence of the friction.

%-------------------------------------------------------
\subsection*{Investment Tax Growth $g_{i,t}$}
%-------------------------------------------------------
The rental rate on capital from the firm FOC is
\begin{align}
  R_{i,t} = \theta^j \frac{P^j_{i,t} Q^j_{i,t}}{K^j_{i,t}} = \theta^j \frac{P^j_{i,t} Y^j_{i,t}}{K^j_{i,t}}, \label{eq:rental}
\end{align}
and the capital stock evolves as
\begin{align}
  K_{i,t+1} = (1 - \delta) K_{i,t} + X_{i,t}, \label{eq:kpath}
\end{align}
From the capital accumulation Euler at $\sigma = 1$ (log utility), the growth rate
of the investment tax is
\begin{align}
  g_{i,t} \equiv \frac{1 - \xi_{i,t+1}}{1 - \xi_{i,t}} =
  \beta R_{i,t+1} \frac{L_{i,t+1}}{L_{i,t}}
  \frac{P^{X}_{i,t}}{P^{X}_{i,t+1}}, \label{eq:xi-recover}
\end{align}
